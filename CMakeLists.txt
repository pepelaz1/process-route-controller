cmake_minimum_required(VERSION 3.20)
project(ProcessRouteController VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(route-tests tests/core_tests.cpp)
target_include_directories(route-tests PRIVATE src)
add_executable(route-report src/report.cpp)
target_include_directories(route-report PRIVATE src)
enable_testing()
add_test(NAME route-core COMMAND route-tests)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(route-world WIN32 src/world_win.cpp)
  target_link_libraries(route-world PRIVATE user32 gdi32)
  add_executable(route-controller src/controller_win.cpp)
  target_link_libraries(route-controller PRIVATE user32)
  add_test(NAME process-integration COMMAND route-controller --integration-test)
  set_tests_properties(process-integration PROPERTIES TIMEOUT 30)
endif()
